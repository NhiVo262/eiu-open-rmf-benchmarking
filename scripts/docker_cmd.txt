docker images --> List images
docker ps -a  --> List containers

docker rmi <image_id>   --> Delete a container
docker rm <container_id> --> Delete an image 

docker compose up --> run containers into docker-compose.yaml
docker compose down  --> Stop & remove containers into docker-compose.yaml

docker compose build --no-cache --> Build images

docker exec -it <container_name> bash --> Access a running container
